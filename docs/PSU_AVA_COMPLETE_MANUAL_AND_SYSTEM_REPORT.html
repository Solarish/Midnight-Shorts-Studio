🏛️ 1. งานเล่าเรื่องประวัติศาสตร์และวัฒนธรรม (Historical & Cultural Heritage)

  • Job 2B (Deep Brain Hologram): สร้างภาพโครงข่ายประสาทสมองพร้อมสกัด Depth Map นำไปทำวงโคจรกล้อง 360 องศาใน AE และซิงก์เสียงกระแสประสาทเข้ากับ Premiere Pro

  • Job 3B (AI Drone Security Patrol): แปลงสไตล์ภาพมุมสูงด้วย comfyui.controlnet_style_transfer (ภาพความร้อน FLIR / แสงกลางคืนสีเขียว NVG) ซ้อนทับ HUD เล็งเป้าหมายอัตโนมัติ

คุณสามารถสร้างเวิร์กโฟลว์ผสมผสาน ComfyUI > After Effects > Premiere Pro ได้อย่างสมบูรณ์แบบ:

import { createGraphDefinition, compileGraphToWorkflow } from "@psu-ava/node-sdk"; const graph = createGraphDefinition({ graphId: "ai_storytelling_master", name: "PSU AI Storytelling — ComfyUI + AE + Premiere Pro", profile: "landscape", nodes: [ { id: "prompt", type: "prompt.scientific_conditioning", config: { subject: "Dental nanobots" } }, { id: "comfy_gen", type: "comfyui.scientific_motion", config: { workflowFile: "workflows/nanobots.json" } }, { id: "ae_vfx", type: "ae.volumetric_particles_3d", config: { particleType: "calcium_ions" } }, { id: "hud", type: "graphics.scientific_hud", config: { title: "TELEMETRY", formula: "Ca10(PO4)6(OH)2" } }, { id: "audio_duck", type: "audio.smart_ducking", config: { targetLufs: -16.0 } }, { id: "export", type: "premiere.export", config: { exports: [{ format: "h264", output: "master.mp4" }] } } ], edges: [ { id: "e1", from: { nodeId: "prompt", port: "positivePrompt" }, to: { nodeId: "comfy_gen", port: "workflowFile" } } ] }); const { workflow } = compileGraphToWorkflow(graph); console.log("Workflow Ready for Automated Multi-Track Assembly:", workflow.id);
